(function () { "use strict"; // Mount the emflow viewer (window.EMFlow) in #project-page-host. // Expects `project_id` and `svc` (the ProjectService implementation) // to be defined by the project_widget_svc*.html template. // // The bundle (emflow.js/css, built in the emflow repo with build_and_copy.sh) // is added to only once per page: this content is often re-loaded // with AJAX (session tabs, workflow editor modal) and the bundle is large. var EMFLOW_JS = {{ url_for_static('libs/js/emflow.js') | tojson }}; var EMFLOW_CSS = {{ url_for_static('libs/css/emflow.css') | tojson }}; function loadEmflow() { if (window.EMFlow) return Promise.resolve(window.EMFlow); if (!window.emflowLoading) { var link = document.createElement("link"); link.rel = "stylesheet"; link.href = EMFLOW_CSS; document.head.appendChild(link); window.emflowLoading = new Promise(function (resolve, reject) { var script = document.createElement("script"); script.src = EMFLOW_JS; script.onload = function () { resolve(window.EMFlow); }; script.onerror = function () { window.emflowLoading = null; reject(new Error("Could not load " + EMFLOW_JS)); }; document.head.appendChild(script); }); } return window.emflowLoading; } function showError(host, err) { host.innerHTML = '
';
    host.firstChild.textContent = "emflow error:\n" + String(err && (err.stack || err));
    console.error(err);
  }

  var host = document.getElementById("project-page-host");

  // The viewer fills the host, so give the host the height left in the
  // window (below the page header, above the card padding and the footer),
  // so the page itself does not scroll. Inside a modal, use a fixed height.
  var MIN_HEIGHT = 450;
  var inModal = $(host).closest(".modal").length > 0;

  function fitHost() {
    if (!document.contains(host))
      return stopFitting();

    if (inModal) {
      host.style.height = "85vh";
      return;
    }
    var rect = host.getBoundingClientRect();
    var footer = document.querySelector(".dashboard-wrapper > .footer");
    var below = footer ? footer.getBoundingClientRect().bottom - rect.bottom : 0;
    var height = Math.max(MIN_HEIGHT,
                          Math.floor(window.innerHeight - (rect.top + window.scrollY) - below));
    if (Math.abs(height - rect.height) > 1)
      host.style.height = height + "px";
  }

  // Content above the host may change (e.g. headers, tabs), so also refit
  // when the page content is resized
  var resizeObserver = window.ResizeObserver ? new ResizeObserver(fitHost) : null;
  if (resizeObserver)
    resizeObserver.observe(document.body);
  window.addEventListener("resize", fitHost);

  function stopFitting() {
    if (resizeObserver)
      resizeObserver.disconnect();
    window.removeEventListener("resize", fitHost);
  }

  fitHost();

  loadEmflow().then(function (EMFlow) {
    var viewer = EMFlow.mountProjectPage({
      container: host,
      service: svc,
      projectName: String(project_id),
    });

    // Inside a modal (workflow editor), stop the viewer when it is closed.
    // Otherwise it is unmounted when the host leaves the page.
    $(host).closest(".modal").one("hidden.bs.modal", function () {
      viewer.unmount();
      stopFitting();
    });
  }).catch(function (err) {
    showError(host, err);
  });
})();
